# Not content. Google had crawled /api/v1/public/download/info and
# /downloads/ethio-build-connect.apk and filed both under "Crawled - currently not
# indexed", which is wasted crawl budget on this site's behalf and noise in the
# coverage report. Uploaded media stays crawlable: listing photos belong in Google
# Images.
Disallow: /api/v1/public/
Disallow: /api/v1/auth/
Disallow: /downloads/
Disallow: /swagger-ui/
Disallow: /v3/api-docs

# Build output, not URLs. /prerender/*.html are the prerendered copies nginx serves for
# /, /exhibition and /ethiopia-real-estate-market; each one names its real URL as
# canonical, but nothing links to these paths and crawling them directly is wasted.
Disallow: /prerender/

# Account and session pages carry a noindex tag too; excluding them here saves the
# crawl entirely rather than fetching them to be told to go away.
Disallow: /login
Disallow: /register
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /go-live
